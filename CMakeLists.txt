cmake_minimum_required(VERSION 3.30)
project(NodeEngine VERSION 0.1.0 LANGUAGES C CXX)
option(NODE_ENGINE_SDL "Build the SDL3 GPU application" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(flecs STATIC deps/flecs-4.1.6/flecs.c)
set_target_properties(flecs PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
target_include_directories(flecs PUBLIC deps/flecs-4.1.6)
target_compile_definitions(flecs PUBLIC FLECS_CUSTOM_BUILD FLECS_OS_API_IMPL)
target_compile_definitions(flecs PRIVATE FLECS_CPP)
if(WIN32)
  target_link_libraries(flecs PUBLIC ws2_32 dbghelp)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(flecs PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
set(NODE_ENGINE_GENERATED "" CACHE FILEPATH "Generated game source")
find_program(CCACHE_PROGRAM ccache)
if(CCACHE_PROGRAM)
  set(CMAKE_CXX_COMPILER_LAUNCHER "${CCACHE_PROGRAM}")
endif()
if(WIN32)
  list(APPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/deps/SDL3-3.4.18/x86_64-w64-mingw32")
  list(APPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/deps/SDL3_mixer-3.2.4/x86_64-w64-mingw32")
  list(APPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/deps/SDL3_image-3.4.8/x86_64-w64-mingw32")
endif()
include(cmake/SdlDependencies.cmake)
node_engine_sdl_dependency()
find_package(SDL3_mixer CONFIG REQUIRED)
find_package(SDL3_image CONFIG REQUIRED)
include(cmake/UiDependencies.cmake)
node_engine_ui_dependencies()
include(cmake/PhysicsDependencies.cmake)
node_engine_physics_dependencies()
include(cmake/AnimationDependencies.cmake)
node_engine_animation_dependencies()
add_custom_target(node_ui_assets
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/fonts"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/deps/RmlUi-6.3/Samples/assets/RobotoMono-Regular.ttf" "${CMAKE_CURRENT_BINARY_DIR}/fonts/RobotoMono-Regular.ttf"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/licenses"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/deps/SDL3-source-3.4.18/SDL3-3.4.18/LICENSE.txt" "${CMAKE_CURRENT_BINARY_DIR}/licenses/SDL3.txt"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/deps/RmlUi-6.3/Samples/assets/LICENSE.txt" "${CMAKE_CURRENT_BINARY_DIR}/licenses/RobotoMono.txt"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/deps/RmlUi-6.3/LICENSE.txt" "${CMAKE_CURRENT_BINARY_DIR}/licenses/RmlUi.txt"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/deps/freetype-2.15.0/docs/FTL.TXT" "${CMAKE_CURRENT_BINARY_DIR}/licenses/FreeType.txt")
add_custom_target(node_audio_assets)
if(WIN32)
  add_custom_command(TARGET node_audio_assets POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL3::SDL3-shared>" "${CMAKE_CURRENT_BINARY_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL3_mixer::SDL3_mixer-shared>" "${CMAKE_CURRENT_BINARY_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL3_image::SDL3_image-shared>" "${CMAKE_CURRENT_BINARY_DIR}")
endif()
if(NODE_ENGINE_SDL)
  set(FASTGLTF_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  add_library(node_display STATIC runtime/src/display.cpp runtime/src/assets.cpp)
  target_link_libraries(node_display PUBLIC node_runtime node_ui_renderer SDL3::SDL3 SDL3_image::SDL3_image-shared fastgltf::fastgltf)
  add_executable(node_view runtime/src/app.cpp)
  target_link_libraries(node_view PRIVATE node_display)
  add_executable(node_gamepad_host_probe EXCLUDE_FROM_ALL tools/gamepad_probe/host.cpp)
  target_link_libraries(node_gamepad_host_probe PRIVATE node_display)
  add_dependencies(node_gamepad_host_probe node_view_assets)
  if(NODE_ENGINE_GENERATED)
    add_executable(node_game_view runtime/src/app.cpp "${NODE_ENGINE_GENERATED}")
    target_compile_definitions(node_game_view PRIVATE NODE_ENGINE_COMPILED=1)
    target_link_libraries(node_game_view PRIVATE node_display)
  endif()
  # Both executables share this output directory. One prerequisite owns staging
  # so parallel link steps cannot write the same shader/DLL files concurrently.
  add_custom_target(node_view_assets
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/runtime/shaders" "${CMAKE_CURRENT_BINARY_DIR}/shaders")
  foreach(target node_view node_game_view)
    if(TARGET ${target})
      set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_BINARY_DIR}>")
      add_dependencies(${target} node_view_assets)
    endif()
  endforeach()
endif()
add_library(node_runtime STATIC runtime/src/values.cpp runtime/src/interpreter.cpp runtime/src/events.cpp runtime/src/scene.cpp runtime/src/prefab_edits.cpp runtime/src/input.cpp runtime/src/materials.cpp runtime/src/sprites.cpp runtime/src/audio.cpp runtime/src/ui.cpp runtime/src/persistence.cpp runtime/src/checkpoint.cpp runtime/src/checkpoint_identity.cpp runtime/src/physics_scene.cpp runtime/generated/adapters.generated.cpp)
include(cmake/CheckpointIdentity.cmake)
add_custom_command(TARGET node_audio_assets POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/licenses"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/runtime/third_party/picosha2/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/licenses/PicoSHA2.txt")
set_source_files_properties(runtime/src/scene.cpp PROPERTIES COMPILE_DEFINITIONS "FLECS_CPP;FLECS_CPP_NO_ENUM_REFLECTION")
target_link_libraries(node_runtime PUBLIC flecs node_ui_session SDL3::SDL3 SDL3_mixer::SDL3_mixer-shared)
target_include_directories(node_runtime PUBLIC runtime/include deps/include deps/glm-1.0.1)
target_sources(node_runtime PRIVATE runtime/src/animation_scene.cpp runtime/src/random.cpp runtime/src/game_time.cpp runtime/src/flow.cpp runtime/src/continuations.cpp runtime/src/input_devices.cpp)
target_sources(node_runtime PRIVATE runtime/src/application.cpp)
target_sources(node_runtime PRIVATE runtime/src/scene_session.cpp)
target_sources(node_runtime PRIVATE runtime/src/node_code.cpp)
target_sources(node_runtime PRIVATE runtime/src/scoped_variables.cpp runtime/src/user_types.cpp)
target_link_libraries(node_runtime PUBLIC node_physics node_animation)
if(MINGW)
  target_link_options(node_runtime INTERFACE -static)
endif()
if(MSVC)
  target_compile_options(node_runtime PUBLIC /utf-8 /W4)
else()
  target_compile_options(node_runtime PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(node_preview runtime/src/headless.cpp)
target_link_libraries(node_preview PRIVATE node_runtime)
add_executable(node_event_probe EXCLUDE_FROM_ALL tools/event_probe/main.cpp)
add_executable(node_gamepad_probe EXCLUDE_FROM_ALL tools/gamepad_probe/main.cpp)
target_link_libraries(node_gamepad_probe PRIVATE node_runtime)
add_executable(node_physics_probe EXCLUDE_FROM_ALL tools/physics_probe/main.cpp)
target_link_libraries(node_physics_probe PRIVATE node_physics)
add_executable(node_character_probe EXCLUDE_FROM_ALL tools/character_probe/main.cpp)
target_link_libraries(node_character_probe PRIVATE node_physics)
add_executable(node_animation_probe EXCLUDE_FROM_ALL tools/animation_probe/main.cpp)
target_include_directories(node_animation_probe PRIVATE deps/include)
target_link_libraries(node_animation_probe PRIVATE node_animation)
target_link_libraries(node_event_probe PRIVATE node_runtime)
add_executable(node_state_probe EXCLUDE_FROM_ALL tools/state_probe/main.cpp)
target_link_libraries(node_state_probe PRIVATE node_runtime)
add_executable(node_game_checkpoint_probe EXCLUDE_FROM_ALL tools/game_checkpoint_probe/main.cpp)
target_link_libraries(node_game_checkpoint_probe PRIVATE node_runtime)
add_executable(node_random_probe EXCLUDE_FROM_ALL tools/random_probe/main.cpp)
target_link_libraries(node_random_probe PRIVATE node_runtime)
add_executable(node_time_probe EXCLUDE_FROM_ALL tools/time_probe/main.cpp)
target_link_libraries(node_time_probe PRIVATE node_runtime)
add_executable(node_flow_probe EXCLUDE_FROM_ALL tools/flow_probe/main.cpp)
target_link_libraries(node_flow_probe PRIVATE node_runtime)
add_executable(node_ui_checkpoint_probe EXCLUDE_FROM_ALL tools/ui_checkpoint_probe/main.cpp)
target_link_libraries(node_ui_checkpoint_probe PRIVATE node_runtime node_ui_renderer)
if(TARGET node_view_assets)
  add_dependencies(node_ui_checkpoint_probe node_view_assets)
endif()
if(NODE_ENGINE_GENERATED)
  add_executable(node_game runtime/src/headless.cpp "${NODE_ENGINE_GENERATED}")
  target_compile_definitions(node_game PRIVATE NODE_ENGINE_COMPILED=1)
  target_link_libraries(node_game PRIVATE node_runtime)
  get_filename_component(node_generated_directory "${NODE_ENGINE_GENERATED}" DIRECTORY)
  set(node_code_source "${node_generated_directory}/node-code.generated.cpp")
  if(EXISTS "${node_code_source}")
    add_library(node_embedded_code OBJECT "${node_code_source}")
    target_link_libraries(node_embedded_code PRIVATE node_runtime)
    foreach(target node_preview node_game node_view node_game_view)
      if(TARGET ${target})
        target_sources(${target} PRIVATE $<TARGET_OBJECTS:node_embedded_code>)
        target_compile_definitions(${target} PRIVATE NODE_ENGINE_NODE_CODE=1)
      endif()
    endforeach()
  endif()
endif()
foreach(target node_preview node_game node_view node_game_view node_event_probe node_gamepad_probe node_gamepad_host_probe node_state_probe node_ui_checkpoint_probe node_game_checkpoint_probe)
  if(TARGET ${target})
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_BINARY_DIR}>")
    add_dependencies(${target} node_audio_assets node_ui_assets)
  endif()
endforeach()
