cmake_minimum_required(VERSION 3.24)
project(NodePhysicsProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Exercise Jolt's internal invariants even with optimized simulation code.
set(USE_ASSERTS ON)
find_program(CCACHE_PROGRAM ccache)
if(CCACHE_PROGRAM)
  set(CMAKE_CXX_COMPILER_LAUNCHER "${CCACHE_PROGRAM}")
endif()
include(../../cmake/PhysicsDependencies.cmake)
node_engine_physics_dependencies()
add_executable(node_physics_probe main.cpp)
target_link_libraries(node_physics_probe PRIVATE node_physics)
add_executable(node_character_probe ../character_probe/main.cpp)
target_link_libraries(node_character_probe PRIVATE node_physics)
